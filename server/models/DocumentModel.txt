import mongoose from "mongoose";

/*
|--------------------------------------------------------------------------
| CHUNK SCHEMA
|--------------------------------------------------------------------------
*/

const chunkSchema = new mongoose.Schema({
  text: {
    type: String,
  },
  chunkIndex: {
    type: Number,
  },
});

/*
|--------------------------------------------------------------------------
| DOCUMENT SCHEMA
|--------------------------------------------------------------------------
*/

const documentSchema = new mongoose.Schema(
  {
    userId: {
      type: mongoose.Schema.Types.ObjectId,
      ref: "User",
      required: true,
    },

    title: {
      type: String,
      required: true,
    },

    fileUrl: {
      type: String,
      required: true,
    },

    fileType: {
      type: String,
      required: true,
    },

    extractedText: {
      type: String,
    },

    chunks: [chunkSchema],
  },
  {
    timestamps: true,
  }
);

/*
|--------------------------------------------------------------------------
| MODEL EXPORT
|--------------------------------------------------------------------------
*/

const DocumentModels = mongoose.model("Document", documentSchema);

export default DocumentModels;
